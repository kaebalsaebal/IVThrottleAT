cmake_minimum_required(VERSION 3.20)
project(ThrottleAT VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(AT_BUILD_ASI "Build the Windows x86 ASI shell" ${WIN32})
add_library(at_core STATIC src/controller.cpp src/config.cpp src/runtime.cpp)
target_include_directories(at_core PUBLIC include)
if(MSVC)
  target_compile_options(at_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(at_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AT_BUILD_ASI)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "ASI requires a 32-bit Windows toolchain; use -A Win32 or the MinGW toolchain file")
  endif()
    if(MSVC)
    add_library(at_minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
      third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde32.c)
    target_include_directories(at_minhook PUBLIC third_party/minhook/include)
    add_library(ThrottleAT SHARED src/plugin.cpp src/ce059.cpp src/x86_gate.cpp src/x86_trace.cpp src/x86_lcp.cpp)
    target_compile_definitions(ThrottleAT PRIVATE AT_CE059=1)
    target_compile_options(ThrottleAT PRIVATE /W4 /permissive- /arch:SSE2)
    target_link_libraries(ThrottleAT PRIVATE at_minhook version user32)
  else()
    message(WARNING "CE059 integration requires MSVC x86. This toolchain builds the inert prototype only.")
    add_library(ThrottleAT SHARED src/plugin.cpp)
  endif()
  target_link_libraries(ThrottleAT PRIVATE at_core)
  set_target_properties(ThrottleAT PROPERTIES PREFIX "" SUFFIX ".asi")
  if(MINGW)
    target_link_options(ThrottleAT PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(at_tests tests/tests.cpp)
  target_link_libraries(at_tests PRIVATE at_core)
  add_test(NAME controller_and_fallback COMMAND at_tests "${CMAKE_CURRENT_SOURCE_DIR}/ThrottleAT.ini")
endif()

if(BUILD_TESTING AND AT_BUILD_ASI)
  add_executable(at_plugin_smoke tests/plugin_smoke.cpp)
  add_dependencies(at_plugin_smoke ThrottleAT)
  add_custom_command(TARGET ThrottleAT POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/ThrottleAT.ini" "$<TARGET_FILE_DIR:ThrottleAT>/ThrottleAT.ini")
  add_test(NAME asi_lifecycle COMMAND at_plugin_smoke "$<TARGET_FILE:ThrottleAT>")
endif()
if(BUILD_TESTING AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4 AND AT_BUILD_ASI)
  add_executable(at_gate_tests tests/gate_tests.cpp src/x86_gate.cpp)
  add_test(NAME x86_gate_cpu_state COMMAND at_gate_tests)
  add_executable(at_ce059_tests tests/ce059_tests.cpp src/x86_gate.cpp src/x86_trace.cpp src/x86_lcp.cpp)
  target_link_libraries(at_ce059_tests PRIVATE at_core at_minhook version user32)
  add_test(NAME ce059_bridge COMMAND at_ce059_tests)
endif()

if(BUILD_TESTING AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4 AND AT_BUILD_ASI)
  add_executable(at_trace_gate_tests tests/trace_gate_tests.cpp src/x86_trace.cpp)
  add_test(NAME x86_trace_cpu_state COMMAND at_trace_gate_tests)
endif()

if(BUILD_TESTING AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4 AND AT_BUILD_ASI)
  add_executable(at_lcp_gate_tests tests/lcp_gate_tests.cpp src/x86_lcp.cpp)
  add_test(NAME x86_lcp_cpu_state COMMAND at_lcp_gate_tests)
endif()

if(BUILD_TESTING)
  add_executable(at_schedule_tests tests/shift_schedule_tests.cpp)
  target_link_libraries(at_schedule_tests PRIVATE at_core)
  add_test(NAME throttle_schedule_scenarios COMMAND at_schedule_tests "${CMAKE_CURRENT_SOURCE_DIR}/ThrottleAT.ini")
endif()
